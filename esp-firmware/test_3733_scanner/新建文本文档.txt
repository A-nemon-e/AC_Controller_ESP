uint8_t i2c_write_reg(const uint8_t addr, const uint8_t reg, const uint8_t *buf,
                      const uint8_t cnt) {
  Wire.beginTransmission(addr);
  Wire.write(reg);
  uint8_t n = Wire.write(buf, cnt);
  Wire.endTransmission();
  return n;
}
uint8_t i2c_read_reg(const uint8_t addr, const uint8_t reg, uint8_t *buf,
                     const uint8_t cnt) {
  Wire.beginTransmission(addr);
  Wire.write(reg);
  Wire.endTransmission();
  uint8_t n = Wire.requestFrom(addr, cnt);
  for (uint8_t i = 0; i < n && i < cnt; i++)
    buf[i] = Wire.read();
  return n;
}

uint8_t fb[SCREEN_ROWS][SCREEN_COLS];
void fbClear() { memset(fb, 0, sizeof(fb)); }
void fbSet(int x, int y, uint8_t v) {
  if (x >= 0 && x < SCREEN_COLS && y >= 0 && y < SCREEN_ROWS)
    fb[y][x] = v;
}
void fbRender() {
  static uint8_t pwmBuf[192];
  for (uint8_t chip = 0; chip < NUM_CHIPS; chip++) {
    memset(pwmBuf, 0, 192);
    for (int8_t sw = CHIP_COLS - 1; sw >= 0; sw--) {
      uint8_t physX = chip * CHIP_COLS + (CHIP_COLS - 1 - sw);
      for (uint8_t cs = 0; cs < SCREEN_ROWS; cs++)
        pwmBuf[sw * 16 + cs] = fb[cs][physX];
    }
    drivers[chip]->SetPWM(pwmBuf);
  }
}
void drawChar(int x, int y, char c, uint8_t bri = 0xFF) {
  int idx = -1;
  if (c >= '0' && c <= '9')
    idx = c - '0';
  else if (c >= 'A' && c <= 'Z')
    idx = 10 + (c - 'A');
  else if (c >= 'a' && c <= 'z')
    idx = 10 + (c - 'a');
  if (idx < 0)
    return;
  for (int r = 0; r < 7; r++) {
    uint8_t row = pgm_read_byte(&FONT5x7[idx][r]);
    for (int col = 0; col < 5; col++)
      if (row & (1 << (4 - col)))
        fbSet(x + col, y + r, bri);
  }
}

bool tryNTPServer(const char *server, uint32_t timeout_ms) {
  configTime(8 * 3600, 0, server);
  uint32_t start = millis();
  while (millis() - start < timeout_ms) {
    time_t now = time(nullptr);
    if (now > 1000000000UL)
      return true;
    delay(100);
    yield();
  }
  return false;
}

void processButton();
void fadeTransition(uint8_t old_fb[SCREEN_ROWS][SCREEN_COLS],
                    uint8_t new_fb[SCREEN_ROWS][SCREEN_COLS]) {
  const int duration_ms = 250;
  uint32_t startMs = millis();
  while (true) {
    uint32_t now = millis();
    int prog = (now - startMs) * 100 / duration_ms;
    if (prog > 100)
      prog = 100;
    for (int y = 0; y < SCREEN_ROWS; y++) {
      for (int x = 0; x < SCREEN_COLS; x++) {
        uint32_t o = old_fb[y][x];
        uint32_t n = new_fb[y][x];
        fb[y][x] = (uint8_t)((o * (100 - prog) + n * prog) / 100);
      }
    }
    fbRender();
    processButton();
    if (currentMode != 3)
      return;
    if (prog >= 100)
      break;
    yield();
  }
  copyBufferToFb(new_fb);
  fbRender();
}
enum NtpState { SHOW_TIME, SHOW_DATE };
uint8_t ntp_current_fb[SCREEN_ROWS][SCREEN_COLS];
uint8_t ntp_next_fb[SCREEN_ROWS][SCREEN_COLS];
void runNTPClock() {
  static uint32_t lastSec = 0xFFFFFFFFUL;
  static NtpState dispState = SHOW_TIME;
  static uint32_t stateStartMs = millis();
  static bool firstFrame = true;
  if (firstFrame) {
    memset(ntp_current_fb, 0, sizeof(ntp_current_fb));
    copyBufferToFb(ntp_current_fb);
    fbRender();
    firstFrame = false;
  }
  if (ntpSynced && millis() - ntpLastSyncMs > NTP_RESYNC_MS) {
    ntpSynced = false;
    setupNTP();
  }
  time_t now = time(nullptr);
  uint32_t sec = (uint32_t)now;
  uint32_t elapsed = millis() - stateStartMs;
  if (dispState == SHOW_TIME && elapsed > 8000) {
    dispState = SHOW_DATE;
    stateStartMs = millis();
  } else if (dispState == SHOW_DATE && elapsed > 2000) {
    dispState = SHOW_TIME;
    stateStartMs = millis();
  }
  static NtpState lastDispState = SHOW_TIME;
  if (sec == lastSec && dispState == lastDispState)
    return;
  lastSec = sec;
  lastDispState = dispState;
  struct tm *ti = localtime(&now);
  memset(ntp_next_fb, 0, sizeof(ntp_next_fb));
  if (!ntpSynced || now < 1000000000UL) {
    drawStrSmallBuffer(ntp_next_fb, 8, 0, "---");
    if (0 + 1 < SCREEN_ROWS)
      ntp_next_fb[0 + 1][20] = 0xFF;
    if (0 + 3 < SCREEN_ROWS)
      ntp_next_fb[0 + 3][20] = 0xFF;

void runMatrixRain() {
  static uint32_t lastFrame = 0;
  if (millis() - lastFrame < 30)
    return;
  lastFrame = millis();
  for (int y = 0; y < SCREEN_ROWS; y++) {
    for (int x = 0; x < SCREEN_COLS; x++) {
      uint8_t v = fb[y][x];
      if (v > 20)
        fb[y][x] = v - 20;
      else
        fb[y][x] = 0;
    }
  }
  for (int x = 0; x < SCREEN_COLS; x++) {
    if (random(100) > 95)
      continue;
    matrixDrops[x] += matrixSpeeds[x];
    int iy = (int)matrixDrops[x];
    if (iy >= 0 && iy < SCREEN_ROWS) {
      fbSet(x, iy, 255);
    }
    if (matrixDrops[x] > SCREEN_ROWS + 5) {
      if (random(100) > 80) {
        matrixDrops[x] = random(-5, 0);
        matrixSpeeds[x] = random(20, 80) / 100.0f;
      }
    }
  }
  fbRender();
}

void runWaterRipples() {
  static float t = 0;
  float cx = SCREEN_COLS / 2.0f - 0.5f;
  float cy = SCREEN_ROWS / 2.0f - 0.5f;
  t += 0.15f;
  if (t > M_PI * 2) {
    t -= M_PI * 2;
  }
  for (int y = 0; y < SCREEN_ROWS; y++) {
    for (int x = 0; x < SCREEN_COLS; x++) {
      float dx = x - cx;
      float dy = y - cy;
      float dist = sqrtf(dx * dx + dy * dy);
      float val = sinf(dist * 0.8f - t);
      uint8_t bri = (uint8_t)((val + 1.0f) * 127.5f);
      fbSet(x, y, bri);
    }
  }
  if (ntpSynced) {
    time_t now = time(nullptr);
    struct tm *ti = localtime(&now);
    drawBigHHMM(ti->tm_hour, ti->tm_min, 0);
  }
  fbRender();
  smartDelay(20);
}

void runAudioBars() {
  static uint32_t lastFrame = 0;
  if (millis() - lastFrame < 40)
    return;
  lastFrame = millis();
  for (int i = 0; i < 6; i++) {
    barCurrentH[i] += (barTargetH[i] - barCurrentH[i]) * 0.15f;
    if (abs(barTargetH[i] - barCurrentH[i]) < 0.5f) {
      if (random(100) > 60)
        barTargetH[i] = random(1, 11);
      else
        barTargetH[i] = 1;
    }
  }
  fbClear();
  for (int y = 0; y < SCREEN_ROWS; y++) {
    float h = SCREEN_ROWS - 1 - y;
    for (int x = 0; x < SCREEN_COLS; x++) {
      float totalBri = 0;
      for (int i = 0; i < 6; i++) {
        if (h <= barCurrentH[i]) {
          float distX = abs(x - barCenters[i]);
          float b = 255.0f * expf(-distX * 0.6f);
          totalBri += b;
        }
      }
      if (totalBri > 255)
        totalBri = 255;
      fbSet(x, y, (uint8_t)totalBri);
    }
  }
  fbRender();
}

void runGaussianBlur() {
  static int state = 0;
  static int blurTicks = 0;
  static uint32_t lastFrame = 0;
  const char *wordA = "ESP8G";
  const char *wordB = "3733";
  if (millis() - lastFrame < 50)
    return;
  lastFrame = millis();
  if (state == 0) {
    fbClear();
    drawStr(centerX(wordA), 2, wordA);
    fbRender();
    smartDelay(2000);
    state = 1;
    blurTicks = 0;
  } else if (state == 2) {
    fbClear();
    drawBigHHMM(12, 34);
    fbRender();
    smartDelay(2000);
    state = 3;
    blurTicks = 0;
  } else if (state == 1 || state == 3) {
    uint8_t temp[SCREEN_ROWS][SCREEN_COLS];
    for (int y = 0; y < SCREEN_ROWS; y++) {
      for (int x = 0; x < SCREEN_COLS; x++) {
        uint32_t sum = 0;
        uint32_t weightSum = 0;
        for (int dy = -1; dy <= 1; dy++) {
          for (int dx = -1; dx <= 1; dx++) {
            int nx = x + dx;
            int ny = y + dy;
            if (nx >= 0 && nx < SCREEN_COLS && ny >= 0 && ny < SCREEN_ROWS) {
              uint8_t weight = 1;
              if (dx == 0 && dy == 0)
                weight = 4;
              else if (dx == 0 || dy == 0)
                weight = 2;
              sum += fb[ny][nx] * weight;
              weightSum += weight;
            }
          }
        }
        temp[y][x] = (uint8_t)(sum / weightSum);
      }
    }
    for (int y = 0; y < SCREEN_ROWS; y++) {
      for (int x = 0; x < SCREEN_COLS; x++) {
        uint8_t v = temp[y][x];
        fb[y][x] = (v > 3) ? (v - 3) : 0;
      }
    }
    fbRender();
    blurTicks++;
    if (blurTicks > 20) {
      if (state == 1)
        state = 2;
      else
        state = 0;
    }
  }
}

void enterBreathMode() {
  Serial.println("[模式1] ABM 同呼吸");
  ABM_CONFIG cfg;
  cfg.T1 = T1_840MS;
  cfg.T2 = T2_210MS;
  cfg.T3 = T3_840MS;
  cfg.T4 = T4_420MS;
  cfg.Tbegin = LOOP_BEGIN_T1;
  cfg.Tend = LOOP_END_T3;
  cfg.Times = (uint16_t)ABM_LOOP_FOREVER;
  for (int i = 0; i < NUM_CHIPS; i++) {
    drivers[i]->Init();
    drivers[i]->WritePagedReg(PAGEDREGISTER::CR, (i == 0) ? 0x41 : 0x81);
    drivers[i]->SetGCC(gccValue);
    drivers[i]->SetLEDMatrixState(LED_STATE::ON);
    drivers[i]->SetLEDMatrixPWM(0xFF);
    drivers[i]->ConfigABM(ABM_NUM::NUM_1, &cfg);
    drivers[i]->SetLEDMatrixMode(LED_MODE::ABM1);
  }
  for (int i = NUM_CHIPS - 1; i >= 0; i--) {
    uint8_t cr = (i == 0) ? 0x43 : 0x83;
    drivers[i]->WritePagedReg(PAGEDREGISTER::CR, cr & ~0x02);
    drivers[i]->WritePagedReg(PAGEDREGISTER::CR, cr);
    drivers[i]->WritePagedReg(PAGEDREGISTER::TUR, 0x00);
    Serial.printf("  chip%d %s CR=0x%02X\n", i, i == 0 ? "MASTER" : "SLAVE",
                  cr);
  }
}

void doLongPressStep() {
  if (!screenOn || _brtBounced)
    return;
  int step = max(1, (int)(8.0f * sqrtf((float)gccValue / 255.0f) + 0.5f));
  int v = (int)gccValue + brtDir * step;
  if (v >= 255) {
    v = 255;
    _brtBounced = true;
  } else if (v <= 1) {
    v = 1;
    _brtBounced = true;
  }
  gccValue = (uint8_t)v;
  applyBrightness();
  Serial.printf("[\u6309\u952e] \u957f\u6309 \u2192 GCC=%d(step=%d)\n",
                gccValue, step);
}
static bool _pendingShort = false;
static uint32_t _pendingMs = 0;
void setupButton() {
  btn.onPressed([]() {
    _pendingShort = true;
    _pendingMs = millis();
  });
  btn.onSequence(2, BTN_DOUBLE_MS, []() {
    if (!screenOn)
      return;
    _pendingShort = false;
    doDoubleClick();
  });
  btn.onPressedFor(BTN_LONG_MS, []() {
    _pendingShort = false;
    if (!screenOn)
      return;
    _brtBounced = false;
    brtDir = -brtDir;
    _brtMs = millis() - BTN_BRT_MS;
    Serial.printf("[按键] 长按?dir=%d\n", brtDir);
  });
  btn.begin();
}
void processButton() {
  btn.read();
  if (_pendingShort && millis() - _pendingMs >= BTN_DOUBLE_MS) {
    _pendingShort = false;
    doShortPress();
  }
  if (screenOn && btn.pressedFor(BTN_LONG_MS)) {
    if (!_brtBounced && millis() - _brtMs >= BTN_BRT_MS) {
      _brtMs = millis();
      doLongPressStep();
    }
  }
  if (btn.wasReleased()) {
    _brtBounced = false;
  }
}
void smartDelay(uint32_t ms) {
  uint32_t start = millis();
  while (millis() - start < ms) {
    processButton();
    yield();
  }
}

    if (currentMode != 2)
      return;
  }
  smartDelay(200);
}
void setup() {
  Serial.begin(115200);
  delay(500);
  pinMode(PIN_SDB, OUTPUT);
  digitalWrite(PIN_SDB, HIGH);
  delay(10);
  setupButton();
  Wire.begin(PIN_SDA, PIN_SCL);
  Wire.setClock(400000);
  delay(50);
  Serial.println("IS31FL3733x4 Tester - ESP-12F");
  Serial.printf(
      "I2C Addr: 0x%02X 0x%02X 0x%02X 0x%02X 0x%02X 0x%02X - Screen %dx%d\n",
      drv0.GetI2CAddress(), drv1.GetI2CAddress(), drv2.GetI2CAddress(),
      drv3.GetI2CAddress(), drv4.GetI2CAddress(), drv5.GetI2CAddress(),
      SCREEN_COLS, SCREEN_ROWS);
  Serial.println("Buttons(IO16): short=power long=dim double=mode");
  WiFi.mode(WIFI_STA);
  WiFi.begin(WIFI_SSID, WIFI_PASS);
  Serial.printf("[WiFi] Connect %s ...\n", WIFI_SSID);
  enterScanMode();
}
void loop() {
  processButton();
  static bool wifiWasConnected = false;
  if (!wifiWasConnected && WiFi.status() == WL_CONNECTED) {
    wifiWasConnected = true;
    Serial.printf("[WiFi] Connected, IP=%s\n",
                  WiFi.localIP().toString().c_str());
    setupNTP();
  }
  if (currentMode == 0) {
    for (uint8_t cs = 0; cs < SCREEN_ROWS; cs++) {
      for (uint8_t chip = 0; chip < NUM_CHIPS; chip++) {
        for (int8_t sw = CHIP_COLS - 1; sw >= 0; sw--) {
          processButton();
          if (currentMode != 0)
            return;
          clearAll();
          drivers[chip]->SetLEDSinglePWM(cs, (uint8_t)sw, 0xFF);
          smartDelay(SCAN_DELAY_MS);
        }
      }
    }
    clearAll();
    smartDelay(300);
  } else if (currentMode == 2) {
    runTextDemo();
  } else if (currentMode == 3) {
    processButton();
    if (currentMode != 3)
      return;
    runNTPClock();
  } else if (currentMode == 4) {
    processButton();
    if (currentMode != 4)
      return;
    runAHTMode();
  } else if (currentMode == 5) {
    processButton();
    if (currentMode != 5)
      return;
    runMatrixRain();
  } else if (currentMode == 6) {
    processButton();
    if (currentMode != 6)
      return;
    runWaterRipples();
  } else if (currentMode == 7) {
    processButton();
    if (currentMode != 7)
      return;
    runAudioBars();
  } else if (currentMode == 8) {
    processButton();
    if (currentMode != 8)
      return;
    runGaussianBlur();
  }
}
